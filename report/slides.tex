\begin{frame}{Short-term PM2.5 Forecasting on the Beijing Multi-Site Dataset}
\vspace{-0.5em}
\centering\footnotesize
\textbf{Ting-Yu Yu} \quad \textbf{Yun-Chen Tsai} \quad $\cdot$ \quad ECE 228 Team 17, UCSD \quad $\cdot$ \quad Spring 2026

\vspace{0.6em}
\begin{columns}[T,onlytextwidth]
% --- left column ---
\begin{column}{0.48\linewidth}
\small
\textbf{Why PM2.5?}\\
Fine particulate matter, deep-lung penetrating, linked to
respiratory and cardiovascular disease.
\alert{$\geq 75\,\mu g/m^3$} is the WHO / Chinese GB
``unhealthy'' threshold -- the level at which cities issue advisories.

\vspace{0.8em}
\textbf{Two questions we set out to answer:}
\begin{enumerate}\setlength\itemsep{0.15em}
  \item Do \alert{sequence models} (LSTM / GRU / Transformer)
        beat feature-based regressors here?
  \item Does \alert{context from the other 11 stations} help?
\end{enumerate}

\vspace{0.8em}
\textbf{Two input settings (our main ablation):}
\begin{itemize}\setlength\itemsep{0.15em}
  \item \alert{global} -- target station's own 24 h history\\
        ($F{=}19$ features per timestep)
  \item \alert{multi-site} -- same, plus channel-wise mean of
        the \emph{other 11 stations} (target masked out)\\
        ($F{=}38$ features per timestep)
\end{itemize}
\end{column}
% --- right column ---
\begin{column}{0.5\linewidth}
\small
\textbf{Data.}\\
UCI Beijing Multi-Site Air Quality:\\
\textbf{12 stations} $\times$ \textbf{hourly} $\times$ \textbf{4 years}
(2013-03 to 2017-02);\\
6 pollutants + 6 weather variables.

\vspace{0.6em}
\textbf{Task.}\\
Sliding window of \alert{24 past hours}
$\Rightarrow$ predict PM2.5 at $t{+}h$ for $h \in \{1, 6\}$ hours.\\
Time-based train / val / test split (no leakage).

\vspace{0.6em}
\textbf{Seven models, one pipeline:}
\begin{itemize}\setlength\itemsep{0.15em}
  \item \textit{No-learning}\,: Persistence $\hat y_{t+h}{=}y_t$
  \item \textit{Feature-based}\,: Ridge, Random Forest, GBM
  \item \textit{Sequence}\,: LSTM, GRU, Transformer encoder
\end{itemize}

\vspace{0.6em}
\textbf{Metric we care most about:}\\
\alert{spike-recall} $=$ fraction of unhealthy hours
($y \geq 75\,\mu g/m^3$) where we also predict $\geq 75$.
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Result 1: GRU multi-site is the best model at both horizons}
\begin{columns}[T,onlytextwidth]
% --- big chart ---
\begin{column}{0.62\linewidth}
\centering
\includegraphics[width=\linewidth]{comparison_MAE.png}\\
{\footnotesize Test MAE per (model, input setting, horizon).
Green/red = multi-site; blue/orange = global. Lower is better.}
\end{column}
% --- numbers + take-aways ---
\begin{column}{0.36\linewidth}
\small
\textbf{1 hour ahead}\\
Persistence is already strong ($R^2 \!\approx\! 0.95$),
but sequence + multi-site pushes further:\\
\textbf{Transformer multi: MAE = 9.41}\\
\textbf{GRU multi: MAE = 9.51}\\
Spike-recall $\approx$ \textbf{0.95} -- we'd flag 19 of every
20 unhealthy hours.

\vspace{0.6em}
\textbf{6 hours ahead}\\
Persistence collapses ($R^2 \!\approx\! 0.56$).
The gap between models opens up:\\
\textbf{GRU multi: MAE = 28.75}\\
GBM multi: 29.00, RF multi: 29.13.

\vspace{0.6em}
\alert{Take-away.} Sequence models pay off most when
paired with the multi-site spatial context.
LSTM under-performs at our default LR/patience -- a
tuning issue, not architecture.
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Result 2: a trivial spatial mean helps -- and the wrap-up}
\begin{columns}[T,onlytextwidth]
% --- delta table ---
\begin{column}{0.39\linewidth}
\small
\textbf{Multi-site setting.} At every timestep, add the channel-wise
mean of the \emph{other 11 stations} ($F: 19 \!\to\! 38$). No graph,
no attention -- the simplest possible spatial summary.

\vspace{0.5em}
\textbf{Test-MAE change vs.\ global} ($\mu g/m^3$, negative = better):
\\[0.2em]
\begin{tabular}{lrr}
\toprule
Model & $\Delta_{h=1}$ & $\Delta_{h=6}$ \\
\midrule
Ridge         & $-0.38$ & $-1.87$ \\
Random Forest & $-0.18$ & $-1.26$ \\
GBM           & $-0.11$ & $-1.43$ \\
GRU           & $-0.19$ & $\mathbf{-2.55}$ \\
Transformer   & $-0.43$ & $-0.79$ \\
\bottomrule
\end{tabular}

\vspace{0.4em}
Consistent across every model; \emph{larger} gains at $h{=}6$,
where local history alone leaves more to predict.
\end{column}
